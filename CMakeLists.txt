cmake_minimum_required(VERSION 3.16)
project(ytdl-gui CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(ytdl-gui WIN32 src/main.cpp)
target_link_libraries(ytdl-gui PRIVATE comctl32 shell32 ole32 user32)
if(MSVC)
  target_compile_options(ytdl-gui PRIVATE /W4 /utf-8)
  target_link_options(ytdl-gui PRIVATE "/MANIFESTDEPENDENCY:type='win32' name='Microsoft.Windows.Common-Controls' version='6.0.0.0' processorArchitecture='*' publicKeyToken='6595b64144ccf1df' language='*'")
endif()
